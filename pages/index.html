# journal_matcher.py - 医学期刊智能匹配引擎
class JournalMatcher:
    """根据研究主题、职称级别、学科方向匹配目标期刊"""
    
    def __init__(self, topic, title_level, department):
        self.topic = topic
        self.title_level = title_level
        self.department = department
        self.journal_db = self._load_journal_database()
    
    def _load_journal_database(self):
        """加载医学期刊数据库（含影响因子、审稿周期等）"""
        return JournalDatabase.load(
            sources=["CNKI期刊目录", "PubMed收录列表", 
                     "中国科技核心期刊目录", "CSCD来源期刊"]
        )
    
    def match(self):
        """执行期刊匹配算法"""
        # 1. 学科方向过滤
        candidates = self.journal_db.filter_by_department(
            self.department)
        # 2. 职称级别权重调整
        weighted = self._apply_title_level_weights(
            candidates, self.title_level)
        # 3. 研究主题相似度计算
        scored = self._compute_topic_similarity(
            weighted, self.topic)
        # 4. 综合评分排序
        results = sorted(scored, key=lambda x: x.score, 
                        reverse=True)
        return results[:3]

# lit_review.py - 文献检索与综述分析引擎
class LiteratureReviewer:
    """多数据库文献检索与空白识别"""
    
    def search(self, topic, databases, year_range):
        """执行多源文献检索"""
        results = {}
        for db in databases:
            if db == "pubmed":
                results[db] = PubMedClient.search(
                    query=self._build_pubmed_query(topic),
                    filters={"date": year_range})
            elif db == "cnki":
                results[db] = CNKIClient.search(
                    query=topic,
                    year_start=year_range[0])
        return self._deduplicate(results)
    
    def identify_gaps(self, literature_set):
        """基于文献计量分析识别研究空白"""
        # 1. 主题聚类分析
        clusters = TopicClusterer.fit(literature_set)
        # 2. 时间趋势分析
        trends = TemporalAnalyzer.analyze(literature_set)
        # 3. 交叉领域空白检测
        gaps = GapDetector.detect(clusters, trends)
        return gaps

# stat_check.py - 统计学方法审查引擎
class StatChecker:
    """医学论文统计方法自动审查"""
    
    def check_methods(self, manuscript):
        """审查论文中使用的统计方法"""
        issues = []
        # 1. 检验方法匹配
        if "t检验" in manuscript and "正态性检验" not in manuscript:
            issues.append(StatIssue(
                level="WARNING",
                message="使用t检验前应先进行正态性检验",
                suggestion="添加Shapiro-Wilk或Kolmogorov-Smirnov检验"))
        # 2. 样本量计算
        if "样本量" not in manuscript:
            issues.append(StatIssue(
                level="ERROR",
                message="缺少样本量计算说明",
                suggestion="补充G*Power或PASS软件的样本量估算"))
        # 3. 多重比较校正
        if manuscript.p_value_count > 3 and "Bonferroni" not in manuscript:
            issues.append(StatIssue(
                level="WARNING",
                message="多次比较需进行校正",
                suggestion="使用Bonferroni或FDR校正方法"))
        return issues

# format_validator.py - 论文格式规范验证引擎
class FormatValidator:
    """医学论文格式合规性验证"""
    
    def validate(self, manuscript, target_journal):
        """根据目标期刊要求验证论文格式"""
        rules = JournalFormatRules.load(target_journal)
        issues = []
        
        # 1. 摘要格式检查
        if not self._check_abstract_format(manuscript, rules):
            issues.append(FormatIssue(
                section="摘要",
                message="摘要字数或格式不符合要求",
                required=f"结构式摘要，{rules.abstract_max_words}字以内"))
        
        # 2. 参考文献格式检查
        ref_issues = self._check_references(
            manuscript.references, rules)
        issues.extend(ref_issues)
        
        # 3. 图表规范检查
        fig_issues = self._check_figures_tables(
            manuscript, rules)
        issues.extend(fig_issues)
        
        return ValidationReport(
            total_issues=len(issues),
            critical=sum(1 for i in issues if i.level=="ERROR"),
            warnings=sum(1 for i in issues if i.level=="WARNING"))
